## 1. Purpose and background

In the confirmation viewer the right-hand list ("Selections:") shows one card
per frame, in the order of SelectionContainer._rects. That same order is the
order of the CSV export and of the Excel keywords (custom_gui/exporter.py
loops over the rects in list order; custom_gui/form_export.py keeps the CSV
order). Nothing sorts by rect_id. Today the only way to change the order is to
delete frames and draw them again. The approver asked (2026-10-09) for an up
and a down button on each card.

Rules that must keep working:
  - rect_id and label NEVER change when a frame moves. Corrected texts
    (self.edits) are keyed by rect_id, and region_id "1" is dropped by the
    Excel conversion (AGENTS.md 8.5). Only the position in the list changes.
  - The existing title row of each card is used by many existing tests:
    card.content is a Column; card.content.controls[0] is a Row with EXACTLY
    two controls [header Text, Row(buttons)]; the tooltips of that button Row
    are asserted as a whole list (tests/test_viewer_task55.py line 84,
    tests/test_viewer_task61.py line 52, tests/test_edit_enter.py line 127);
    card.content.controls[1] is the text / edit area. DO NOT add anything to
    the title row or to the button Row, and do not change controls[0] or [1].
  - The button Row is already full: 300 px list, up to 7 icons of 40 px,
    the delete icon touches the right border. So the new buttons go on a
    THIRD line of the card (card.content.controls[2]), small, right-aligned.

## 2. Contract

### DoD-1 -- custom_gui/selection.py: one new method on SelectionContainer,
placed just before delete_by_id. Exactly this:

    def move_by(self, rect_id: str, delta: int) -> bool:
        """Move one place up (-1) or down (+1) in the list. ids and labels do not change."""
        if delta not in (-1, 1):
            return False
        for i, rect in enumerate(self._rects):
            if rect.rect_id == rect_id:
                j = i + delta
                if j < 0 or j >= len(self._rects):
                    return False
                self._rects[i], self._rects[j] = self._rects[j], self._rects[i]
                return True
        return False

Nothing else in selection.py changes (_next_id is not touched).

### DoD-2 -- custom_gui/app.py, SelectableImageViewer

(a) New method, placed just before `def start_region_ocr(`. Exactly this:

    def move_region(self, rid, delta: int) -> None:

            self._update_status()
            return
        if not self.selection_container.move_by(rid, delta):

def get_prepared_app(dummy_page, tmp_path):
    app, path = get_app(dummy_page, tmp_path, make_clean_page)
    app.ocr_results = []
    app.selection_container.restore([])
    for bbox in [(0, 0, 60, 30), (80, 80, 200, 200), (10, 300, 90, 400)]:
        app.selection_container.add(bbox)
    # ids "1","2","3"; drop "1" so the rows are Region 2, 3, 4 as in the viewer
    app.selection_container.delete_by_id("1")
    app.selection_container.add((5, 500, 50, 550))
    app.edits = {"2": "二", "3": "三", "4": "四"}
    app.image_states[str(path)]["edits"] = app.edits
    app.editing_region_id = None
    app._update_selections_ui()
    return app, str(path)


def ids(app):
    return [r.rect_id for r in app.selection_container.get_all()]


def move_buttons(row):
    move_row = row.content.controls[2]
    return move_row.controls[0], move_row.controls[1]


def click(btn):
    class E:
        control = btn
    btn.on_click(E())


def test_1_move_by_contract():
    c = SelectionContainer()
    for i in range(3):
        c.add((i, i, i + 1, i + 1))
    before_next = c.add((9, 9, 10, 10)).rect_id   # "4"
    c.delete_by_id("4")
    assert [r.rect_id for r in c.get_all()] == ["1", "2", "3"]

    assert c.move_by("2", -1) is True
    assert [r.rect_id for r in c.get_all()] == ["2", "1", "3"]
    assert [r.label for r in c.get_all()] == ["Region 2", "Region 1", "Region 3"]

    assert c.move_by("2", -1) is False          # already first
    assert c.move_by("3", 1) is False           # already last
    assert c.move_by("9", 1) is False           # unknown id
    assert c.move_by("1", 2) is False           # only -1 / +1
    assert c.move_by("1", 0) is False
    assert [r.rect_id for r in c.get_all()] == ["2", "1", "3"]

    assert c.move_by("1", 1) is True
    assert [r.rect_id for r in c.get_all()] == ["2", "3", "1"]
    assert c.add((0, 0, 1, 1)).rect_id == str(int(before_next) + 1)   # numbering untouched


def test_2_buttons_place_and_size(dummy_page, tmp_path):
    app, _ = get_prepared_app(dummy_page, tmp_path)
    for row in app.selections_list.controls:

            ["Edit", "右から変換", "この枠だけOCR", "画像の枠を隠す", "縦で読む", "Revert to OCR", "Delete"]

        assert isinstance(move_row, ft.Row)
        assert len(move_row.controls) == 2
        up, down = move_row.controls
        assert (up.icon, up.tooltip) == (ft.Icons.ARROW_UPWARD, "上へ")
        assert (down.icon, down.tooltip) == (ft.Icons.ARROW_DOWNWARD, "下へ")
        for b in (up, down):
            assert b.icon_size == 18 and b.width == 32 and b.height == 32
        assert move_row.alignment == ft.MainAxisAlignment.END


def test_3_down_moves_row_and_keeps_text(dummy_page, tmp_path):
